import React from 'react';
import { BrowserRouter, NavLink, Route, Routes } from 'react-router-dom';
import Login from './login/login';
import Play from './play/game';
import Leaderboard from './leaderboard/leaderboard';
import About from './about/about';
import 'bootstrap/dist/css/bootstrap.min.css';
import './App.css';

function App() {
  // Manage authentication state
  const [userName, setUserName] = React.useState(localStorage.getItem('userName') || '');
  const isLoggedIn = !!userName; // Boolean indicating login status

  // Handle login - update state and localStorage
  const handleLogin = (name) => {
    setUserName(name);
    localStorage.setItem('userName', name);
  };

  // Handle logout - clear state and localStorage
  const handleLogout = () => {
    setUserName('');
    localStorage.removeItem('userName');
  };

  return (
    <BrowserRouter>
      <div className="body bg-light text-dark">
        {/* Navbar with conditional links */}
        <header className="container-fluid">
          <nav className="navbar navbar-light bg-light">
            <div className="container-fluid d-flex justify-content-between">
              <NavLink className="navbar-brand" to="/about">
                <h1 className="h3">In A Blink</h1>
              </NavLink>
              <ul className="nav">
                {!isLoggedIn && (
                  <li className="nav-item">
                    <NavLink className="nav-link" to="/">Login/Register</NavLink>
                  </li>
                )}
                <li className="nav-item">
                  <NavLink className="nav-link" to="/about">About</NavLink>
                </li>
                {isLoggedIn && (
                  <>
                    <li className="nav-item">
                      <NavLink className="nav-link" to="/play">Play</NavLink>
                    </li>
                    <li className="nav-item">
                      <NavLink className="nav-link" to="/leaderboard">Leaderboard</NavLink>
                    </li>
                    <li className="nav-item">
                      <button className="btn btn-secondary ms-3" onClick={handleLogout}>Logout</button>
                    </li>
                  </>
                )}
              </ul>
            </div>
          </nav>
        </header>

        {/* Routes */}
        <Routes>
          {/* Redirect to Login if not logged in */}
          <Route
            path="/"
            element={
              isLoggedIn ? <Navigate to="/play" /> : <Login onLogin={handleLogin} />
            }
          />
          <Route path="/about" element={<About />} />
          <Route path="/play" element={isLoggedIn ? <Play userName={userName} /> : <Navigate to="/" />} />
          <Route path="/leaderboard" element={isLoggedIn ? <Leaderboard /> : <Navigate to="/" />} />
          <Route path="*" element={<Navigate to="/" />} /> {/* Redirect undefined routes to Login */}
        </Routes>
        <footer className="footer bg-light text-black-50 py-1 mt-3">
            <div className="container text-center">
                <span className="text-reset">Callie Lambourne</span>
                <a href="https://github.com/callielamb/startup.git" className="ms-2 text-decoration-none">GitHub</a>
            </div>
        </footer>
      </div>
    </BrowserRouter>
  );
}

export default App;
